Raspberry Pi Zero 2W: Driving AHT20 Sensor and Displaying Temperature & Humidity on OLED

2026-04-20
简体中文

This article walks through how to drive an AHT20 temperature & humidity sensor with a Raspberry Pi Zero 2 W using Python, and display the readings on a 0.96-inch OLED screen (SSD1306/CH1116) with large fonts and a humidity progress bar. The project uses Python 3 with the smbus2 and Pillow libraries, with code generated with the help of Gemini AI.

Prerequisites

Item Description
Raspberry Pi Zero 2 W Running Raspberry Pi OS
AHT20 module Temperature & humidity sensor, I2C interface
0.96" OLED SSD1306 / CH1116 driver, I2C interface
Jumper wires Female-to-male, 4 pieces
Software Python 3, smbus2, Pillow, WenQuanYi Zen Hei font

Hardware Connection

Both the AHT20 and OLED use the I2C interface. Since the Raspberry Pi’s I2C bus supports multiple devices, simply connect them in parallel.

Module Pin Description Raspberry Pi Pin (Physical) Function
VCC Power (3.3V) Pin 1 3.3V Power
GND Ground Pin 9 Ground
SCL I2C Clock Pin 5 GPIO 3 (SCL)
SDA I2C Data Pin 3 GPIO 2 (SDA)

Usage

  1. Make sure I2C is enabled (see Software Setup)
  2. Activate the virtual environment: source myenv/bin/activate
  3. Run the script: python3 oled_aht20.py
  4. The OLED will refresh temperature and humidity readings every 2 seconds

import time
import os
from smbus2 import SMBus
from PIL import Image, ImageDraw, ImageFont

# --- Configuration ---
I2C_BUS = 1
AHT20_ADDR = 0x38
OLED_ADDR = 0x3C


# --- Font Loading ---
def get_font(size):
    """Try multiple common paths to find a Chinese font"""
    font_paths = [
        "simhei.ttf",
        "/usr/share/fonts/truetype/wqy/wqy-zenhei.ttc",
        "/usr/share/fonts/opentype/noto/NotoSansCJK-Regular.ttc"
    ]
    for path in font_paths:
        if os.path.exists(path):
            return ImageFont.truetype(path, size)

    print(f"Warning: Chinese font not found, using default (size: {size})")
    return ImageFont.load_default()


# --- OLED Driver ---
def oled_command(bus, cmd):
    bus.write_byte_data(OLED_ADDR, 0x00, cmd)


def oled_init(bus):
    """Initialize OLED with charge pump and page addressing mode"""
    cmds = [0xAE, 0xD5, 0x80, 0xA8, 0x3F, 0xD3, 0x00, 0x40, 0x8D, 0x14,
            0x20, 0x02, 0xA1, 0xC8, 0xDA, 0x12, 0x81, 0xCF, 0xD9, 0xF1,
            0xDB, 0x40, 0xA4, 0xA6, 0xAF]
    for c in cmds:
        oled_command(bus, c)


def oled_display(bus, image):
    """Push a PIL Image to the OLED"""
    img = image.convert("1")
    pix = img.load()
    for page in range(8):
        oled_command(bus, 0xB0 + page)
        oled_command(bus, 0x00)
        oled_command(bus, 0x10)
        page_data = []
        for x in range(128):
            byte = 0
            for bit in range(8):
                if pix[x, page * 8 + bit]:
                    byte |= (1 << bit)
            page_data.append(byte)
        for i in range(0, 128, 32):
            bus.write_i2c_block_data(OLED_ADDR, 0x40, page_data[i:i + 32])


# --- AHT20 Sensor ---
def read_aht20(bus):
    """Read AHT20, returns (temperature in °C, humidity in %)"""
    bus.write_i2c_block_data(AHT20_ADDR, 0xAC, [0x33, 0x00])
    time.sleep(0.08)
    data = bus.read_i2c_block_data(AHT20_ADDR, 0x71, 7)
    h_raw = ((data[1] << 12) | (data[2] << 4) | (data[3] >> 4))
    t_raw = ((data[3] & 0x0F) << 16) | (data[4] << 8) | data[5]
    return (t_raw / 1048576 * 200 - 50), (h_raw / 1048576 * 100)


# --- Main Program ---
def main():
    bus = SMBus(I2C_BUS)
    oled_init(bus)

    # Load three font sizes
    f_huge = get_font(30)   # Temperature digits
    f_mid = get_font(16)    # Labels
    f_small = get_font(12)  # Unit symbols

    while True:
        try:
            temp, humi = read_aht20(bus)
            img = Image.new("1", (128, 64))
            draw = ImageDraw.Draw(img)

            # --- Temperature display ---
            t_str = f"{temp:.1f}"
            draw.text((5, -2), t_str, font=f_huge, fill=255)
            t_w = draw.textlength(t_str, font=f_huge)
            draw.text((5 + t_w + 2, 12), "°C", font=f_mid, fill=255)

            # --- Separator line ---
            draw.line((0, 31, 128, 31), fill=255)

            # --- Humidity display ---
            draw.text((5, 35), f"Humidity:", font=f_mid, fill=255)
            draw.text((50, 35), f"{humi:.1f}%", font=f_mid, fill=255)

            # --- Humidity progress bar ---
            draw.rectangle((5, 54, 122, 60), outline=255, fill=0)
            bar_w = int(humi / 100 * 115)
            draw.rectangle((7, 56, 7 + bar_w, 58), outline=255, fill=255)

            oled_display(bus, img)
        except Exception as e:
            print(f"Error: {e}")

        time.sleep(2)


if __name__ == "__main__":
    main()
